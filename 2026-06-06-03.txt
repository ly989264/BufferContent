tag="DMSG$(date +%s)"

sudo iptables -I OUTPUT 1 -p icmp -d 127.0.0.1 \
  -j LOG --log-prefix "$tag:" --log-level 6 || exit 1

ping -c 1 -W 1 127.0.0.1 >/dev/null 2>&1 || true

sudo iptables -D OUTPUT -p icmp -d 127.0.0.1 \
  -j LOG --log-prefix "$tag:" --log-level 6 2>/dev/null || true

dmesg -T | grep "$tag" | tail -n 5
